cmake_minimum_required(VERSION 3.16)
project(ConFiGifY LANGUAGES C CXX)
include(GNUInstallDirs)

option(BUILD_SHARED_LIBS "Build shared libraries for language bindings" ON)
add_library(cfgy src/cpp/cfgy.cpp src/cpp/evaluation.cpp src/c/cfgy_c.cpp)
target_include_directories(cfgy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src/cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/c")
target_compile_features(cfgy PUBLIC cxx_std_17)
set_target_properties(cfgy PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON POSITION_INDEPENDENT_CODE ON)
install(TARGETS cfgy)
install(FILES src/c/cfgy.h src/cpp/cfgy DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")

add_executable(example examples/cpp/example.cpp)
target_link_libraries(example PRIVATE cfgy)

add_executable(example-c examples/c/example.c)
target_compile_features(example-c PRIVATE c_std_11)
target_link_libraries(example-c PRIVATE cfgy)

option(CFGY_JAVA "Build the Java jar and JNI module" OFF)
if(CFGY_JAVA)
    find_package(Java REQUIRED COMPONENTS Development)
    find_package(JNI REQUIRED)
    include(UseJava)
    add_library(cfgy_jni SHARED src/java/jni.c)
    target_include_directories(cfgy_jni PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(cfgy_jni PRIVATE cfgy)
    add_jar(cfgy_java src/java/io/configify/Cfgy.java OUTPUT_NAME cfgy)
endif()

option(CFGY_LUA "Build the Lua module" OFF)
if(CFGY_LUA)
    find_package(Lua 5.3 REQUIRED)
    add_library(cfgy_lua MODULE src/lua/cfgy_lua.c)
    target_include_directories(cfgy_lua PRIVATE ${LUA_INCLUDE_DIR})
    target_link_libraries(cfgy_lua PRIVATE cfgy ${LUA_LIBRARIES})
    set_target_properties(cfgy_lua PROPERTIES PREFIX "" OUTPUT_NAME cfgy
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lua"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lua" WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()

option(CFGY_MATLAB "Build the MATLAB MEX module" OFF)
if(CFGY_MATLAB)
    find_package(Matlab REQUIRED)
    matlab_add_mex(NAME cfgy_native SRC src/matlab/native.c LINK_TO cfgy)
endif()

option(BUILD_TESTING "Build the C++ parser tests" OFF)
include(CTest)
if(BUILD_TESTING)
    add_executable(cfgy-c-test tests/c/test.c)
    target_compile_features(cfgy-c-test PRIVATE c_std_11)
    target_link_libraries(cfgy-c-test PRIVATE cfgy)
    add_test(NAME cfgy-c-api COMMAND cfgy-c-test "${CMAKE_CURRENT_SOURCE_DIR}/tests/conformance.cfgy")
    add_executable(cfgy-test tests/cpp/test_configify.cpp)
    target_link_libraries(cfgy-test PRIVATE cfgy)
    add_test(NAME cfgy-parser COMMAND cfgy-test)
    set_tests_properties(cfgy-parser PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME cfgy-example-cli COMMAND example "${CMAKE_CURRENT_SOURCE_DIR}/examples/example.cfgy")
    add_test(NAME cfgy-example-c-cli COMMAND example-c "${CMAKE_CURRENT_SOURCE_DIR}/examples/example.cfgy")
    set_tests_properties(cfgy-example-cli cfgy-example-c-cli PROPERTIES PASS_REGULAR_EXPRESSION "800[\r\n]+2[.]5[\r\n]")
endif()
